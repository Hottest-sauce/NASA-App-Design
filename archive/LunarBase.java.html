public class LunarBase
{
    private String baseName;
    private double waterSupply;
    private double oxygenSupply;
    public LunarBase(String a)
    {
        baseName = a;
        waterSupply = 100;
        oxygenSupply = 100;
    }
    public String getBaseName()
    {
        return baseName;
    }

    public double getWaterSupply()
    {
        return waterSupply;
    }
    public double getOxygenSupply()
    {
        return oxygenSupply;
    }

    public String toString()
    {
        return new String("Water Supply: " + waterSupply + " Oxygen Supply: " + oxygenSupply );
    }



}
